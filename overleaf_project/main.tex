\newcommand{\makechaptertitlepage}[2]{%
\cleardoublepage
\thispagestyle{fancy}

\vspace*{\fill}

\begin{center}
    {\Large\scshape\bfseries\color{blue!85!black} CHAPTER #1}\\[0.25in]

    {\color{gray!40}\rule{4.0in}{1.2pt}}\\[0.25in]

    {\Huge\bfseries\color{black}
    \parbox{0.82\textwidth}{%
        \centering
        \setstretch{1.1}%
        #2
    }}\\[0.25in]

    {\color{gray!40}\rule{4.0in}{1.2pt}}
\end{center}

\vspace*{\fill}
\newpage
}

          \centering
          \vfill

          {\textbf{#1}}
          \vfill

      }%
    }%
